function init() {
  scene = new THREE.Scene();
  camera = new THREE.PerspectiveCamera(45, innerWidth/innerHeight, 0.1, 1000);

  renderer.setSize(innerWidth, innerHeight);
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  document.getElementById('canvas-container').appendChild(renderer.domElement);

  // 全亮照明
  scene.add(new THREE.AmbientLight(0xffffff, 1.5));
  const sun1 = new THREE.DirectionalLight(0xffffff, 0.75);
  sun1.position.set(5,3,5); scene.add(sun1);
  const sun2 = new THREE.DirectionalLight(0xffffff, 0.45);

function onResize() {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
}

function createStars() {
  const geo = new THREE.BufferGeometry();
  const pos = [];
  for (let i=0;i<4000;i++) pos.push((Math.random()-0.5)*1600,(Math.random()-0.5)*1600,(Math.random()-0.5)*1600);
  geo.setAttribute('position', new THREE.Float32BufferAttribute(pos,3));
  stars = new THREE.Points(geo, new THREE.PointsMaterial({color:0xffffff, size:1.2}));
  scene.add(stars);
}

function createEarth() {
  const geo = new THREE.SphereGeometry(R, 64, 64);
  const map = new THREE.TextureLoader().load('https://unpkg.com/three-globe/example/img/earth-blue-marble.jpg');
  earth = new THREE.Mesh(geo, new THREE.MeshBasicMaterial({map}));
  scene.add(earth);
}

function createAtmosphere() {
  const geo = new THREE.SphereGeometry(R*1.015, 64, 64);
  atmosphere = new THREE.Mesh(geo, new THREE.ShaderMaterial({
    vertexShader:`varying vec3 vNormal;void main(){vNormal=normalize(normalMatrix*normal);gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.);}`,
    fragmentShader:`varying vec3 vNormal;void main(){float i=pow(0.68-dot(vNormal,vec3(0,0,1.)),2.0);gl_FragColor=vec4(0.3,0.6,1.0,1.0)*i;}`,
    blending:THREE.AdditiveBlending,side:THREE.BackSide,transparent:true
  }));
  scene.add(atmosphere);
}

function latLngToVec(lat, lng, radius = R + 0.04) {
  const phi = (90-lat)*Math.PI/180;
  const theta = (lng+180)*Math.PI/180;
  return new THREE.Vector3(
    -radius*Math.sin(phi)*Math.cos(theta),
     radius*Math.cos(phi),
     radius*Math.sin(phi)*Math.sin(theta)
  );
}

function createPermanentTenDash() {
  tenDashGroup = new THREE.Group();
  TEN_DASH.forEach(seg => {
    const pts = seg.map(([lng,lat]) => latLngToVec(lat, lng, R + 0.028));
    const line = new THREE.Line(
      new THREE.BufferGeometry().setFromPoints(pts),
      new THREE.LineBasicMaterial({ color: 0xff0000, linewidth: 2 })
    );
    tenDashGroup.add(line);
  });
  scene.add(tenDashGroup);
}

async function loadWorldBorders() {
  worldBorders = new THREE.Group();
  scene.add(worldBorders);
  try {
    const res = await fetch('https://raw.githubusercontent.com/nvkelso/natural-earth-vector/master/geojson/ne_110m_admin_0_countries.geojson');
    const gj = await res.json();
    window._worldFeatures = gj.features;
    gj.features.forEach(f => {
      const n = (f.properties.NAME || f.properties.ADMIN || '').toLowerCase();
      if (n.includes('china')) return;
      const coords = f.geometry.type === 'Polygon' ? [f.geometry.coordinates] : f.geometry.coordinates;
      coords.forEach(poly => poly.forEach(ring => {
        const pts = ring.map(([lng,lat]) => latLngToVec(lat,lng,R+0.012));
        const line = new THREE.Line(new THREE.BufferGeometry().setFromPoints(pts),
          new THREE.LineBasicMaterial({color:0x64748b,transparent:true,opacity:0.35}));
        worldBorders.add(line);
      }));
    });
  } catch(e){console.warn(e);}
}

function createPin(color) {
  const g = new THREE.Group();
  const cone = new THREE.Mesh(new THREE.ConeGeometry(0.03,0.09,8), new THREE.MeshBasicMaterial({color}));
  cone.rotation.x = Math.PI; cone.position.y = -0.045;
  g.add(cone);
  const head = new THREE.Mesh(new THREE.SphereGeometry(0.04,10,10), new THREE.MeshBasicMaterial({color}));
  head.position.y = 0.02;
  g.add(head);
  return g;
}

function createLabelSprite(text, color = '#ffffff') {
  const canvas = document.createElement('canvas');
  canvas.width = 256; canvas.height = 48;
  const ctx = canvas.getContext('2d');
  ctx.fillStyle = 'rgba(0,0,0,0.72)';
  ctx.beginPath();
  if (ctx.roundRect) ctx.roundRect(0,0,256,48,6); else ctx.rect(0,0,256,48);
  ctx.fill();
  ctx.fillStyle = color;
  ctx.font = 'bold 18px -apple-system, sans-serif';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText(text.substring(0,16), 128, 24);
  const texture = new THREE.CanvasTexture(canvas);
  texture.needsUpdate = true;
  const mat = new THREE.SpriteMaterial({ map: texture, transparent: true, depthTest: false });
  const sprite = new THREE.Sprite(mat);
  sprite.scale.set(0.65, 0.13, 1);
  return sprite;
}

window.addPointFromInput = function() {
  const input = document.getElementById('searchPlace').value.trim();
  const name = document.getElementById('pointName').value.trim();
  const color = document.getElementById('pointColor').value;
  if (!input) return alert('請輸入地名或座標');

  const parts = input.split(/[,，\s]+/).map(s => parseFloat(s)).filter(n => !isNaN(n));
  if (parts.length >= 2) {
    addPoint(name || `${parts[0].toFixed(4)},${parts[1].toFixed(4)}`, parts[0], parts[1], color);
    return;
  }

  fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(input)}&limit=1`, {
    headers: {'User-Agent': 'EarthPro/7.0'}
  }).then(r => r.json()).then(data => {
    if (!data.length) return alert('找不到地點');
    addPoint(name || data[0].display_name.split(',')[0], parseFloat(data[0].lat), parseFloat(data[0].lon), color);
  }).catch(() => alert('搜尋失敗，請檢查網路'));
};

function addPoint(name, lat, lng, color) {
  const pos = latLngToVec(lat, lng);
  const pin = createPin(color);
  pin.position.copy(pos);
  pin.lookAt(pos.clone().multiplyScalar(2));

  const sprite = createLabelSprite(name);
  sprite.position.copy(pos.clone().multiplyScalar(1.075));

  const group = new THREE.Group();
  group.add(pin);
  group.add(sprite);
  scene.add(group);

  points.push({id: nextId++, name, lat, lng, color, group, pin, sprite, type: 'point'});
  saveCurrentProject();
  renderPoints();
  updatePathSelects();
  document.getElementById('searchPlace').value = '';
  document.getElementById('pointName').value = '';
}

function renderPoints() {
  document.getElementById('pointCount').textContent = points.length;
  const list = document.getElementById('pointList');
  list.innerHTML = '';
  points.forEach((p, i) => {
    const div = document.createElement('div');
    div.className = 'item';
    div.style.borderLeftColor = p.color;
    div.innerHTML = `<div class="name"><span class="color-dot" style="background:${p.color}"></span>${p.name}</div>
      <small>${p.lat.toFixed(5)}, ${p.lng.toFixed(5)}</small>
      <div class="item-actions">
        <button class="small secondary" onclick="flyToPoint(${i})">飛去</button>

      </div>`;
    list.appendChild(div);
  });
}

window.toggleEditPoint = i => document.getElementById(`edit-point-${i}`).classList.toggle('show');
window.saveEditPoint = i => {
  const p = points[i];
  p.name = document.getElementById(`edit-pname-${i}`).value.trim() || p.name;
  p.color = document.getElementById(`edit-pcolor-${i}`).value;
  p.pin.traverse(c => { if (c.isMesh) c.material.color.set(p.color); });
  // 更新文字
  const newSprite = createLabelSprite(p.name);
  newSprite.position.copy(p.sprite.position);
  p.group.remove(p.sprite);
  p.group.add(newSprite);
  p.sprite = newSprite;
  saveCurrentProject();
  renderPoints();
  updatePathSelects();
};

window.flyToPoint = function(i) {
  const p = points[i];
  const target = latLngToVec(p.lat, p.lng, R + 2.3);
  const start = camera.position.clone();
  const t0 = performance.now();
  function anim(t) {
    const k = Math.min((t - t0) / 750, 1);
    const e = k < 0.5 ? 2*k*k : -1 + (4-2*k)*k;
    camera.position.lerpVectors(start, target, e);
    controls.target.set(0,0,0);
    if (k < 1) requestAnimationFrame(anim);
  }
  requestAnimationFrame(anim);
};

window.removePoint = i => {
  scene.remove(points[i].group);
  points.splice(i,1);
  saveCurrentProject();
  renderPoints();
  updatePathSelects();
};

window.clearPoints = () => {
  if (!confirm('清除所有坐標點？')) return;
  points.forEach(p => scene.remove(p.group));
  points = [];
  saveCurrentProject();
  renderPoints();
  updatePathSelects();
};

window.addCountryBorder = function() {
  const displayName = document.getElementById('countryName').value.trim() || document.getElementById('countryKey').value.trim();
  const key = document.getElementById('countryKey').value.trim() || displayName;
  const style = document.getElementById('countryStyle').value;
  const color = document.getElementById('countryColor').value;
  if (!key) return alert('請輸入國家關鍵字');

  const group = new THREE.Group();
  let centerLat = 35, centerLng = 105;
  let found = false;

  if (key.includes('中國') || key.toLowerCase().includes('china') || key.includes('中国')) {
    const outlinePts = CHINA_CLAIM.map(([lng,lat]) => latLngToVec(lat, lng, R+0.02));
    const outline = new THREE.LineLoop(new THREE.BufferGeometry().setFromPoints(outlinePts),
      new THREE.LineBasicMaterial({color, transparent:true, opacity:0.95}));
    group.add(outline);
    if (style === 'fill') {
      const fillMat = new THREE.MeshBasicMaterial({color, transparent:true, opacity:0.22, side:THREE.DoubleSide});
      const fillGeo = new THREE.SphereGeometry(R+0.014, 32, 16, 0, Math.PI*2, 0.28, 1.15);
      const fillMesh = new THREE.Mesh(fillGeo, fillMat);
      fillMesh.rotation.y = -0.75;
      group.add(fillMesh);
    }
    centerLat = 35; centerLng = 105;
    found = true;
  } else if (window._worldFeatures) {
    window._worldFeatures.forEach(f => {
      const n = (f.properties.NAME || f.properties.ADMIN || '').toLowerCase();
      if (n.includes(key.toLowerCase()) || key.toLowerCase().includes(n)) {
        let sumLat=0, sumLng=0, cnt=0;
        const coords = f.geometry.type === 'Polygon' ? [f.geometry.coordinates] : f.geometry.coordinates;
        coords.forEach(poly => poly.forEach(ring => {
          const pts = ring.map(([lng,lat]) => {
            sumLat += lat; sumLng += lng; cnt++;
            return latLngToVec(lat, lng, R+0.017);
          });
          const line = new THREE.Line(new THREE.BufferGeometry().setFromPoints(pts),
            new THREE.LineBasicMaterial({color, transparent:true, opacity:0.9}));
          group.add(line);
        }));
        if (cnt > 0) { centerLat = sumLat/cnt; centerLng = sumLng/cnt; }
        found = true;
      }
    });
  }

  if (!found) return alert('找不到該國家');

  const sprite = createLabelSprite(displayName, '#ffffff');
  sprite.position.copy(latLngToVec(centerLat, centerLng, R + 0.18));
  sprite.scale.set(1.1, 0.22, 1);
  group.add(sprite);

  scene.add(group);
  countries.push({
    id: nextId++, name: displayName, key, color, style, group,
    lat: centerLat, lng: centerLng, type: 'country', sprite
  });
  saveCurrentProject();
  renderCountries();
  updatePathSelects();
  document.getElementById('countryName').value = '';
  document.getElementById('countryKey').value = '';
};

function updatePathSelects() {
  const from = document.getElementById('pathFrom');
  const to = document.getElementById('pathTo');
  from.innerHTML = '<option value="">起點</option>';
  to.innerHTML = '<option value="">終點</option>';
  points.forEach((p,i) => {
    from.innerHTML += `<option value="point_${i}">📍 ${p.name}</option>`;
    to.innerHTML += `<option value="point_${i}">📍 ${p.name}</option>`;
  });
  countries.forEach((c,i) => {
    from.innerHTML += `<option value="country_${i}">🏳️ ${c.name}</option>`;
    to.innerHTML += `<option value="country_${i}">🏳️ ${c.name}</option>`;

}

window.addPath = function() {
  const fromVal = document.getElementById('pathFrom').value;
  const toVal = document.getElementById('pathTo').value;
  if (!fromVal || !toVal || fromVal === toVal) return alert('請選擇不同起點與終點');

  let a, b;
  if (fromVal.startsWith('point_')) a = points[parseInt(fromVal.split('_')[1])];
  else a = countries[parseInt(fromVal.split('_')[1])];
  if (toVal.startsWith('point_')) b = points[parseInt(toVal.split('_')[1])];
  else b = countries[parseInt(toVal.split('_')[1])];

  const name = document.getElementById('pathName').value.trim() || `路徑${paths.length+1}`;
  const color = document.getElementById('pathColor').value;

  const pts = [];
  for (let i=0; i<=64; i++) {
    const t = i/64;
    const lat = a.lat + (b.lat - a.lat)*t;
    const lng = a.lng + (b.lng - a.lng)*t;
    pts.push(latLngToVec(lat, lng, R+0.028));
  }
  const line = new THREE.Line(new THREE.BufferGeometry().setFromPoints(pts), new THREE.LineBasicMaterial({color}));
  scene.add(line);
  const particle = new THREE.Mesh(new THREE.SphereGeometry(0.025,8,8), new THREE.MeshBasicMaterial({color}));
  scene.add(particle);

  paths.push({id:nextId++, name, color, line, particle, points:pts, t:0});
  saveCurrentProject();
  renderPaths();
  document.getElementById('pathName').value = '';
};

function renderPaths() {
  document.getElementById('pathCount').textContent = paths.length;
  const list = document.getElementById('pathList');
  list.innerHTML = '';
  paths.forEach((p,i) => {
    const div = document.createElement('div');
    div.className = 'item';
    div.style.borderLeftColor = p.color;
    div.innerHTML = `<div class="name"><span class="color-dot" style="background:${p.color}"></span>${p.name}</div>
      <div class="item-actions"><button class="small danger" onclick="removePath(${i})">刪除</button></div>`;
    list.appendChild(div);
  });
}

window.removePath = i => {
  scene.remove(paths[i].line);
  scene.remove(paths[i].particle);
  paths.splice(i,1);
  saveCurrentProject();
  renderPaths();
};

window.clearPaths = () => {

  paths = [];
  saveCurrentProject();
  renderPaths();
};

window.startTour = function() {
  if (points.length === 0) return alert('請先新增坐標點');
  isTouring = true; isPaused = false; tourIdx = 0;
  document.getElementById('tourBar').classList.add('show');
  playTour();
};
function playTour() {
  if (!isTouring || isPaused) return;
  flyToPoint(tourIdx);
  document.getElementById('tourStatus').textContent = `${tourIdx+1}/${points.length} ${points[tourIdx].name}`;
  const delay = (parseFloat(document.getElementById('tourDelay').value) || 4) * 1000;
  const once = document.getElementById('tourOnce').checked;
  tourTimer = setTimeout(() => {
    tourIdx++;
    if (tourIdx >= points.length) {
      if (once) { stopTour(); return; }
      tourIdx = 0;
    }
    playTour();
  }, delay);
}
window.pauseTour = () => { isPaused = !isPaused; if (!isPaused) playTour(); };
window.stopTour = () => { isTouring = false; clearTimeout(tourTimer); document.getElementById('tourBar').classList.remove('show'); };
window.nextTour = () => { if (isTouring) { clearTimeout(tourTimer); tourIdx = (tourIdx+1)%points.length; playTour(); } };
window.prevTour = () => { if (isTouring) { clearTimeout(tourTimer); tourIdx = (tourIdx-1+points.length)%points.length; playTour(); } };

function loadAllProjects() {
  try {
    const raw = localStorage.getItem('earth_projects_v7');
    projects = raw ? JSON.parse(raw) : {'預設專案': {points:[], countries:[], paths:[]}};
  } catch(e) { projects = {'預設專案': {points:[], countries:[], paths:[]}}; }
  if (!projects[currentProject]) projects[currentProject] = {points:[], countries:[], paths:[]};
  refreshProjectSelect();
}
function refreshProjectSelect() {
  const sel = document.getElementById('projectSelect');
  sel.innerHTML = '';
  Object.keys(projects).forEach(name => {
    const opt = document.createElement('option');
    opt.value = name; opt.textContent = name;
    if (name === currentProject) opt.selected = true;
    sel.appendChild(opt);
  });
}
window.createProject = function() {
  const name = document.getElementById('newProjectName').value.trim();
  if (!name) return alert('請輸入名稱');
  if (projects[name]) return alert('已存在');
  projects[name] = {points:[], countries:[], paths:[]};
  currentProject = name;
  saveAllProjects();
  refreshProjectSelect();
  clearSceneData();
  document.getElementById('newProjectName').value = '';
};
window.switchProject = function() {
  currentProject = document.getElementById('projectSelect').value;
  clearSceneData();
  const data = projects[currentProject] || {points:[], countries:[], paths:[]};
  (data.points || []).forEach(p => addPoint(p.name, p.lat, p.lng, p.color));
  renderPoints(); renderCountries(); renderPaths();
  updatePathSelects();
};
window.deleteProject = function() {
  if (Object.keys(projects).length <= 1) return alert('至少保留一個');
  if (!confirm(`刪除「${currentProject}」？`)) return;
  delete projects[currentProject];
  currentProject = Object.keys(projects)[0];
  saveAllProjects();
  refreshProjectSelect();
  switchProject();
};
function clearSceneData() {
  points.forEach(p => scene.remove(p.group));
  countries.forEach(c => scene.remove(c.group));
  paths.forEach(p => { scene.remove(p.line); scene.remove(p.particle); });
  points = []; countries = []; paths = [];
}
function saveCurrentProject() {
  projects[currentProject] = {
    points: points.map(p => ({name:p.name, lat:p.lat, lng:p.lng, color:p.color})),
    countries: countries.map(c => ({name:c.name, key:c.key, color:c.color, style:c.style, lat:c.lat, lng:c.lng})),
    paths: paths.map(p => ({name:p.name, color:p.color}))
  };
  saveAllProjects();
}
function saveAllProjects() {
  try { localStorage.setItem('earth_projects_v7', JSON.stringify(projects)); } catch(e){}
}

function animate() {
  requestAnimationFrame(animate);
  controls.update();

  // 路徑流動
  paths.forEach(p => {
    p.t = (p.t + 0.006) % 1;
    const idx = Math.floor(p.t * (p.points.length - 1));
    p.particle.position.copy(p.points[idx]);
  });

  // 文字與 Pin 隨距離大幅動態縮放（解決放大後文字過大遮蓋問題）
  const dist = camera.position.length();
  const scale = Math.max(0.18, Math.min(1.35, 4.8 / dist));
  points.forEach(p => {
    p.pin.scale.setScalar(scale);
    if (p.sprite) p.sprite.scale.set(0.65 * scale, 0.13 * scale, 1);
  });
  countries.forEach(c => {
    if (c.sprite) {
      const cScale = Math.max(0.4, Math.min(1.6, 6 / dist));
      c.sprite.scale.set(1.1 * cScale, 0.22 * cScale, 1);
    }
  });

  renderer.render(scene, camera);
}
